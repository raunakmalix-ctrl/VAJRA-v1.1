# Main Colab env — Gradio + SDXL + face swap + faster-whisper.
# torch / torchvision / torchaudio are preinstalled by Colab; do NOT pin them
# here or you risk breaking CUDA. Everything below layers on top.

# gradio 4.44.x is the last line that imports HfFolder (needs huggingface_hub<1.0).
gradio==4.44.1
gradio-client==1.3.0
# gradio 4.44 breaks on newer starlette (TemplateResponse -> "unhashable type:
# 'dict'"); pin the fastapi/starlette pair it was built against.
fastapi==0.112.2
starlette<0.39

# SDXL (text -> image). Pinned as ONE mutually consistent set with
# huggingface_hub 0.25.2 below — Colab's newer transformers/diffusers expect
# huggingface_hub>=1.5 (is_offline_mode), which conflicts with gradio 4.x.
diffusers==0.31.0
transformers==4.44.2
tokenizers<0.20
accelerate>=0.34.0
safetensors
sentencepiece
protobuf

# Face swap
insightface==0.7.3
# onnxruntime-gpu is NOT pinned here on purpose. It has to match the host's
# CUDA (newer releases want CUDA 13 and fail with "libcudart.so.13: cannot
# open shared object file", which breaks face swap AND SDXL loading via
# diffusers' onnx_utils probe), so the right build can only be chosen at
# install time -- install_main.sh does that, and falls back to the CPU
# provider rather than failing.
#
# It used to be pinned here to an exact version. When that version was later
# delisted from PyPI, pip refused the WHOLE file -- so gradio, torch and
# everything else silently went uninstalled and the app ran against whatever
# the ambient environment happened to have. One unavailable wheel must not be
# able to take out the entire install.
opencv-python-headless
gfpgan
basicsr
facexlib
lpips        # CodeFormer's basicsr imports this

# Transcript (Feature 2, step 1) — faster-whisper gives word timestamps without
# WhisperX's pyannote/torchaudio coupling (which breaks on Colab's torchaudio).
faster-whisper

# DSP realism layer (dsp/): BS.1770 loudness for splice matching. Pure
# numpy/scipy, no heavy deps -- degrades to RMS matching if unavailable.
pyloudnorm

# Media utils
pydub
soundfile
imageio-ffmpeg
# Media Studio: background removal (downloads u2net on first use). rembg pulls
# onnxruntime (CPU); install_main.sh re-pins onnxruntime-gpu afterwards.
rembg
# Exact pin: has HfFolder (for gradio) and works with diffusers 0.31 / transformers 4.44.
huggingface_hub==0.25.2

# insightface / basicsr need numpy < 2
numpy<2
